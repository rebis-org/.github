\def\textfieldx#1#2#3#4#5#6#7{%
  \pdfextension annot width #2 height #3 depth 1pt {%
    /Subtype /Widget /FT /Tx /T (#1) /V (#4) /F 4 /Ff #5 /Q 0 /H /I
    /AP << /N #7\space 0 R >>
    /BS << /W 0.5 >>
    /MK << /BC [0.45 0.45 0.45] /BG [#6] >>
  }%
  \edef\lastfieldno{\the\numexpr\pdffeedback lastannot\relax}%
  \edef\act{\global\noexpand\fieldtoks={\the\fieldtoks\lastfieldno\space 0 R }}%
  \act
}

\def\formfield#1#2{%
  \iffiled
    \hbox to #1{\filedbox{#1}{#2}\hss}%
  \else
    \ifarchival
      \hbox to #1{\staticfield{#1}{10mm}\hss}%
    \else
      \ifdim#1=76mm
        \hbox to #1{\textfieldhalf{#2}{#1}{10mm}\hss}%
      \else
        \hbox to #1{\textfield{#2}{#1}{10mm}\hss}%
      \fi
    \fi
  \fi
}
